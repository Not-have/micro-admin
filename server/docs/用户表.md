# 用户表

## 1. 创建表

```sql
CREATE TABLE `admin_user` (
  `user_id` INT UNSIGNED NOT NULL AUTO_INCREMENT COMMENT '用户ID',
  `username` VARCHAR(50) NOT NULL COMMENT '用户名',
  `password` VARCHAR(20) NOT NULL COMMENT '密码',
  `real_name` VARCHAR(50) DEFAULT NULL COMMENT '真实姓名',
  `gender` ENUM('M','F','U') DEFAULT 'U' COMMENT '性别 M男 F女 U未知',
  `mobile` VARCHAR(20) DEFAULT NULL COMMENT '手机号',
  `email` VARCHAR(100) DEFAULT NULL COMMENT '邮箱',
  `department_id` INT DEFAULT NULL COMMENT '所属部门ID',
  `role_type` ENUM('superadmin','admin','staff') NOT NULL DEFAULT 'staff' COMMENT '角色类型',
  `user_status` ENUM('active','disabled','resigned') NOT NULL DEFAULT 'active' COMMENT '用户状态',
  `create_time` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `update_time` DATETIME DEFAULT NULL ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  `last_login` DATETIME DEFAULT NULL COMMENT '最后登录时间',
  `is_deleted` TINYINT(1) NOT NULL DEFAULT '0' COMMENT '删除标记 0正常 1删除',
  PRIMARY KEY (`user_id`),
  UNIQUE KEY `uniq_username` (`username`),
  UNIQUE KEY `uniq_mobile` (`mobile`),
  UNIQUE KEY `uniq_email` (`email`),
  KEY `idx_department` (`department_id`),
  KEY `idx_status` (`user_status`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='后台用户信息表';
```
